Step 1: Create Your Web Application Folder

bash
Copy

Step 2: Initialize Git in Your Project Folder
bash
Copy
Edit
git init
This command creates a .git folder — this is your local Git repository.

Step 3: Configure Git (if not already configured)
bash
Copy
Edit
git config --global user.name "Your Name"
git config --global user.email "your.email@example.com"
Step 4: Add Files to the Staging Area
bash
Copy
Edit
git add .
This adds all files in your project directory to Git's staging area.

You can also use git add index.html to add specific files.

Step 5: Commit the Files
bash
Copy
Edit
git commit -m "Initial commit - added index.html"
This saves the snapshot of your project history.

// Generator function for infinite Fibonacci sequence
function* fibonacciGenerator() {
  let a = 0, b = 1;

  while (true) {
    yield a;
    [a, b] = [b, a + b]; // Swap and update
  }
}

// Create generator object
const fib = fibonacciGenerator();

// Print first 10 Fibonacci numbers
for (let i = 0; i < 10; i++) {
  console.log(fib.next().value);

const server = http.createServer((req, res) => {
  if (req.url === '/') {
    res.writeHead(200, { 'Content-Type': 'text/html' });
    res.end('<h2>Hello ramu! This is HTML</h2>');

  else if (req.url === '/text') {
    res.writeHead(200, { 'Content-Type': 'text/plain' });
    res.end('Just plain text here, ramu!');

  else if (req.url === '/js') {
    res.writeHead(200, { 'Content-Type': 'application/javascript' });
    res.end('console.log("Hello from JavaScript, ramu!");');
  }

  else {
    res.writeHead(404, { 'Content-Type': 'text/plain' });
    res.end('404 Not Found');
  }
});

function factorial(n) {
  if (n === 0 || n === 1) return 1;
  return n * factorial(n - 1);
}

module.exports = factorial; // 👈 Default export

// fibonacci.js

function fibonacci(n) {
  const series = [0, 1];
  for (let i = 2; i < n; i++) {
    series[i] = series[i - 1] + series[i - 2];
  }
  return series.slice(0, n); // returns first n numbers
}

module.exports = fibonacci; // 👈 Default export

function loginUser(username, password) {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (username === 'ramu' && password === '1234') {
        resolve('🎉 Login successful!');
      } else {
        reject('❌ Invalid credentials');

  });
}

// app.js

const express = require('express');
const mongoose = require('mongoose');
const User = require('./models/User');

const app = express();
app.use(express.json());

// 🔗 Connect to MongoDB
mongoose.connect('mongodb://127.0.0.1:27017/userDB', {
  useNewUrlParser: true,
  useUnifiedTopology: true
}).then(() => console.log('✅ MongoDB connected'))
  .catch(err => console.error('❌ DB Connection Error:', err));

// ✅ CREATE User
app.post('/users', async (req, res) => {
  try {
    const user = new User(req.body);
    const saved = await user.save();
    res.status(201).json(saved);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// 📄 READ all Users
app.get('/users', async (req, res) => {
  try {
    const users = await User.find();
    res.json(users);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// 📝 UPDATE User by ID
app.put('/users/:id', async (req, res) => {
  try {
    const updated = await User.findByIdAndUpdate(req.params.id, req.body, { new: true });
    res.json(updated);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// ❌ DELETE User by ID
app.delete('/users/:id', async (req, res) => {
  try {
    await User.findByIdAndDelete(req.params.id);
    res.json({ message: 'User deleted successfully' });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// Start server
app.listen(3000, () => {
  console.log('🚀 Server running at http://localhost:3000');
});
